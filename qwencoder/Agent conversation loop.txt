"""Agent conversation loop.

Derived from hermes-agent's AIAgent.run_conversation() pattern:
- Synchronous tool-use loop with max_iterations budget
- OpenAI-format messages (system/user/assistant/tool)
- Tool dispatch through the unified registry
- Context compression when approaching token limits

Extended with:
- Provider routing (select best free LLM per call)
- Permission-aware tool dispatch
- Execution audit trail for VAP chain
"""

from __future__ import annotations

import json
import logging
import time
from dataclasses import dataclass, field
from typing import Any, Optional

from ..tools.registry import ToolRegistry
from ..tools.permissions import PermissionContext
from ..providers.router import ProviderRouter, RouteResult

logger = logging.getLogger(__name__)


@dataclass
class ConversationResult:
    final_response: str
    messages: list[dict[str, Any]]
    tool_calls_count: int
    iterations: int
    provider_used: Optional[RouteResult] = None
    duration_ms: float = 0.0


class AgentLoop:
    """Core agent conversation loop.

    Combines:
    - hermes-agent's sequential tool-use loop
    - Provider routing for model selection
    - Permission-gated tool dispatch
    - nexusalpha TokenGuard budget tracking
    """

    def __init__(
        self,
        tool_registry: Optional[ToolRegistry] = None,
        provider_router: Optional[ProviderRouter] = None,
        permission_ctx: Optional[PermissionContext] = None,
        max_iterations: int = 90,
    ) -> None:
        self.tool_registry = tool_registry or ToolRegistry()
        self.provider_router = provider_router
        self.permission_ctx = permission_ctx
        self.max_iterations = max_iterations

    def run(
        self,
        user_message: str,
        system_message: Optional[str] = None,
        conversation_history: Optional[list[dict]] = None,
    ) -> ConversationResult:
        """Run the agent conversation loop.

        This is a synchronous loop that mirrors hermes-agent's
        run_conversation() pattern. For actual LLM calls, it uses
        the provider router to select the best free provider.
        """
        start = time.monotonic()
        messages: list[dict[str, Any]] = []

        if system_message:
            messages.append({"role": "system", "content": system_message})

        if conversation_history:
            messages.extend(conversation_history)

        messages.append({"role": "user", "content": user_message})

        tool_schemas = self.tool_registry.get_definitions(
            permission_ctx=self.permission_ctx,
        )

        route_result = None
        if self.provider_router:
            route_result = self.provider_router.select(
                min_context=32_000,
                prefer="quality",
                fallback_chain=True,
            )

        iterations = 0
        tool_calls_count = 0

        # In a real implementation, this loop would call the LLM API.
        # Here we demonstrate the architecture and dispatch pattern.
        # The actual LLM client would use route_result.provider.base_url
        # and route_result.model.id for the API call.

        duration = (time.monotonic() - start) * 1000

        return ConversationResult(
            final_response=f"[AgentLoop ready — {len(tool_schemas)} tools available, "
                          f"provider={route_result.provider.name if route_result else 'none'}, "
                          f"model={route_result.model.name if route_result else 'none'}]",
            messages=messages,
            tool_calls_count=tool_calls_count,
            iterations=iterations,
            provider_used=route_result,
            duration_ms=round(duration, 2),
        )
