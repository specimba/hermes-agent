"""Governance hooks for NEXUS OS integration.

These hooks allow the unified platform to consult nexusalpha's Governor
before executing tools, routing to providers, or running workflows.

Architecture:
- GovernanceHooks is the integration point (import nexus_os.governor when available)
- GovernanceDecision wraps allow/deny with reason and trust score
- AuditEntry records actions for VAP chain integration

From nexusalpha:
- Governor.check(action) → allowed/denied
- TrustScorer v2.1 → lane-scoped Bayesian reputation
- TokenGuard → budget enforcement
"""

from __future__ import annotations

import hashlib
import threading
import time
from dataclasses import dataclass, field
from enum import Enum
from typing import Any, Optional


class DecisionType(Enum):
    ALLOWED = "allowed"
    DENIED = "denied"
    HOLD = "hold"         # GSPP pending approval


@dataclass(frozen=True)
class GovernanceDecision:
    decision: DecisionType
    reason: str
    trust_score: float = 1.0
    lane: str = "general"
    token_budget_remaining: Optional[int] = None


@dataclass
class AuditEntry:
    """VAP-chain-compatible audit entry.

    Uses SHA-256 hashing for immutable trail,
    matching nexusalpha's L1 (proposal) + L2 (merge proof) pattern.
    """
    timestamp: float
    action: str
    actor: str
    decision: DecisionType
    details: dict[str, Any] = field(default_factory=dict)
    parent_hash: str = ""

    @property
    def entry_hash(self) -> str:
        import json as _json
        details_str = _json.dumps(self.details, sort_keys=True, default=str)
        content = f"{self.timestamp}:{self.action}:{self.actor}:{self.decision.value}:{self.parent_hash}:{details_str}"
        return hashlib.sha256(content.encode()).hexdigest()[:16]


class GovernanceHooks:
    """Integration hooks for nexusalpha governance.

    When nexusalpha is available (imported), all checks go through
    the real Governor. Otherwise, defaults to permissive mode with
    audit logging.
    """

    def __init__(self, strict: bool = True) -> None:
        self.strict = strict
        self._audit_log: list[AuditEntry] = []
        self._token_budget: int = 100_000
        self._tokens_used: int = 0
        self._governor = None
        self._db = None
        self._db_path: Optional[str] = None
        self._lock = threading.Lock()

        try:
            from nexus_os.governor import NexusGovernor
            from nexus_os.db.manager import DatabaseManager, DBConfig
            import tempfile
            import os
            import uuid as _uuid
            self._db_path = os.path.join(
                tempfile.gettempdir(),
                f"nexus_unified_governor_{os.getpid()}_{_uuid.uuid4().hex[:8]}.db",
            )
            passphrase = os.environ.get("NEXUS_GOVERNOR_PASSPHRASE", "")
            _val = os.environ.get("NEXUS_UNIFIED_ALLOW_UNENCRYPTED", "").lower()
            allow_unencrypted = _val in ("1", "true", "yes")
            encrypted = bool(passphrase) and not allow_unencrypted
            config = DBConfig(
                db_path=self._db_path, passphrase=passphrase,
                encrypted=encrypted,
                allow_unencrypted=allow_unencrypted or not passphrase,
            )
            self._db = DatabaseManager(config=config)
            self._db.setup_schema()
            self._governor = NexusGovernor(db=self._db)
        except (ImportError, Exception):
            pass

    def close(self) -> None:
        """Close the governor DB connection and clean up temp file."""
        if self._db is not None:
            try:
                self._db.close()
            except Exception:
                pass
            self._db = None
        if self._db_path is not None:
            import os
            try:
                os.unlink(self._db_path)
            except OSError:
                pass
            self._db_path = None
        self._governor = None

    def __enter__(self) -> "GovernanceHooks":
        return self

    def __exit__(self, *exc: Any) -> None:
        self.close()

    def check_tool_access(self, tool_name: str, actor: str = "agent") -> GovernanceDecision:
        """Check if a tool call is permitted."""
        if self._governor:
            try:
                result = self._governor.check_access(
                    agent_id=actor,
                    project_id="nexus-unified",
                    action=tool_name,
                )
                decision_map = {
                    "allow": DecisionType.ALLOWED,
                    "deny": DecisionType.DENIED,
                    "hold": DecisionType.HOLD,
                }
                mapped = decision_map.get(result.decision.value, DecisionType.DENIED)
                decision = GovernanceDecision(
                    decision=mapped,
                    reason=f"Governor: {result.decision.value}",
                )
                self._record(tool_name, actor, decision.decision)
                return decision
            except Exception:
                if self.strict:
                    decision = GovernanceDecision(
                        decision=DecisionType.DENIED,
                        reason="Governor error; strict mode denies by default",
                    )
                    self._record(tool_name, actor, decision.decision)
                    return decision
                decision = GovernanceDecision(
                    decision=DecisionType.ALLOWED,
                    reason="Governor error; fail-open (non-strict)",
                )
                self._record(tool_name, actor, decision.decision)
                return decision

        if self.strict:
            decision = GovernanceDecision(
                decision=DecisionType.DENIED,
                reason="No governor available; strict mode denies by default",
            )
        else:
            decision = GovernanceDecision(
                decision=DecisionType.ALLOWED,
                reason="No governor available; default permissive (non-strict)",
            )
        self._record(tool_name, actor, decision.decision)
        return decision

    def check_token_budget(self, tokens_requested: int) -> GovernanceDecision:
        """TokenGuard integration — check if token budget allows the request."""
        with self._lock:
            remaining = self._token_budget - self._tokens_used

        if remaining <= 0:
            return GovernanceDecision(
                decision=DecisionType.DENIED,
                reason="Token budget exhausted (hard stop)",
                token_budget_remaining=0,
            )

        if tokens_requested > remaining:
            return GovernanceDecision(
                decision=DecisionType.DENIED,
                reason=f"Request ({tokens_requested}) exceeds remaining budget ({remaining})",
                token_budget_remaining=remaining,
            )

        return GovernanceDecision(
            decision=DecisionType.ALLOWED,
            reason="Within budget",
            token_budget_remaining=remaining,
        )

    def track_tokens(self, tokens_used: int, model: str = "", actor: str = "agent") -> None:
        """Track token usage (nexusalpha TokenTracker pattern)."""
        if tokens_used < 0:
            raise ValueError(f"tokens_used must be non-negative, got {tokens_used}")
        with self._lock:
            self._tokens_used += tokens_used
        self._record(
            action=f"token_usage:{tokens_used}:{model}",
            actor=actor,
            decision=DecisionType.ALLOWED,
        )

    def _record(self, action: str, actor: str, decision: DecisionType) -> None:
        with self._lock:
            parent = self._audit_log[-1].entry_hash if self._audit_log else ""
            entry = AuditEntry(
                timestamp=time.time(),
                action=action,
                actor=actor,
                decision=decision,
                parent_hash=parent,
            )
            self._audit_log.append(entry)

    @property
    def audit_log(self) -> list[AuditEntry]:
        with self._lock:
            return list(self._audit_log)

    @property
    def tokens_remaining(self) -> int:
        with self._lock:
            return max(0, self._token_budget - self._tokens_used)

    def set_token_budget(self, budget: int) -> None:
        with self._lock:
            self._token_budget = budget
